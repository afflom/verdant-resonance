definition{
  name: breaksAcousticHeadroom,
  parameters: [gainMilli : Nat],
  result: Bool,
  body: ((gainMilli =? (verdantMasterGainMilli)) && ((verdantLimiterCeilingMilli) <? gainMilli))
}

theorem{
  name: acousticHeadroomStandardHolds,
  parameters: [gainMilli : Nat],
  statement: ((breaksAcousticHeadroom gainMilli) = false),
  proof: linear_arithmetic{definitions: [breaksAcousticHeadroom, verdantLimiterCeilingMilli, verdantMasterGainMilli]},
  axioms: [Quot.sound, propext]
}
